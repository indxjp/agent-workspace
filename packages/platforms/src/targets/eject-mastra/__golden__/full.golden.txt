=== README.md ===
# support-triage
Triages customer support tickets and drafts replies.

## Run

```sh
npm install
cp .env.example .env   # then fill in the values
node --env-file=.env src/main.ts "your prompt"
```

## Code tools

Copy your project's `tools/` directory next to `src/` and add any package your run files import.
`src/code-tools.ts` loads the declarations from it.

## Mastra

The agent is a Mastra `Agent`. `src/agent.ts` exports `createAgent()`, which holds the workspace tools each permission enables, the MCP servers, the skills and the subagents; `.mastra/skills/` holds the skills.
The model is routed by Mastra from `<provider>/<id>`; an agent that names none runs on `anthropic/claude-sonnet-4-5`. Set the key of each provider in `.env`.
A remote MCP server is reached over streamable HTTP, with a fall back to SSE.

## What was not carried over

The platform report that came with this project gives every field a verdict and says why a field could not be carried. An ejected project is not importable back into Agent Workspace.

import { fileURLToPath } from "node:url";
import path from "node:path";
import { loadCodeTools } from "./code-tools.ts";

const projectRoot = path.resolve(path.dirname(fileURLToPath(import.meta.url)), "..");

function requireEnv(name: string): string {
  const value = process.env[name];
  if (value === undefined || value === "") {
    throw new Error("Set " + name + " (see .env.example)");
  }
  return value;
}

const workspaces: Workspace[] = [];

function workspaceFor(
  id: string,
  tools: WorkspaceToolsConfig,
  skills: readonly string[],
  commands: boolean,
): Workspace {
  const workspace = new Workspace({
    id,
    filesystem: new LocalFilesystem({ basePath: projectRoot }),
    ...(commands ? { sandbox: new LocalSandbox({ workingDirectory: projectRoot }) } : {}),
    ...(skills.length === 0 ? {} : { skills: [...skills] }),
    tools,
  });
  workspaces.push(workspace);
  return workspace;
}

export async function createAgent(): Promise<{ agent: Agent; close(): Promise<void> }> {

  const codeToolsFor = (names: readonly string[]) =>

          codeTool.name,
          createTool({
            id: codeTool.name,
            description: codeTool.description,
            inputSchema: codeTool.input,
            execute: async (input) => codeTool.call(input),

  const mcp = new MCPClient({ id: "support-triage", servers: {

    await mcp.disconnect();

  const mcpTools = (names: readonly string[]) =>
    Object.fromEntries(
      names.flatMap((server) =>
        Object.entries(toolsets[server] ?? {}).map(([name, tool]) => [server + "_" + name, tool]),

      enabled: false,
      mastra_workspace_read_file: {
        enabled: true,
      },
      mastra_workspace_list_files: {
        enabled: true,
      },
      mastra_workspace_grep: {
        enabled: true,
      },
      mastra_workspace_file_stat: {
        enabled: true,
      },
    }, [".mastra/skills/refund-policy"], false),
  });
  const agent = new Agent({
    id: "support-triage",
    name: "support-triage",
    description: "Triages customer support tickets and drafts replies.",
    instructions: "You triage support tickets.\n\n- Look up the order before answering.\n- Hand billing questions to billing-specialist.",
    model: "anthropic/claude-sonnet-4-6",

        enabled: true,

      },
      mastra_workspace_file_stat: {
        enabled: true,

      maxSteps: 20,

=== src/code-tools.ts ===
import { randomUUID } from 'node:crypto';
import { readdir, realpath } from 'node:fs/promises';
import path from 'node:path';
import { pathToFileURL } from 'node:url';

import { isCodeTool } from '@indx-workspace/sdk/define';
import { z } from 'zod';

export interface LoadedCodeTool {
  readonly name: string;
  readonly description: string;
  readonly input: z.ZodObject;
  call(input: unknown): Promise<unknown>;
}

interface RunContext {
  readonly signal: AbortSignal;
  readonly runId: string;
  readonly agent: string;
}

type RunFunction = (input: unknown, context: RunContext) => unknown;

const runId = randomUUID();

async function filesBelow(directory: string): Promise<string[]> {
  const entries = await readdir(directory, { withFileTypes: true }).catch(() => []);
  const nested = await Promise.all(
    entries.map(async (entry) => {
      const file = path.join(directory, entry.name);
      if (entry.isDirectory()) return filesBelow(file);
      return /(?<!\.d)\.[jt]s$/u.test(entry.name) ? [file] : [];
    }),
  );
  return nested.flat();
}

async function defaultExport(file: string): Promise<unknown> {
  return ((await import(pathToFileURL(file).href)) as { default?: unknown }).default;
}

async function runFile(root: string, run: string): Promise<string> {
  const real = await realpath(path.join(root, ...run.split('/')));
  const relative = path.relative(await realpath(root), real);
  if (relative.startsWith('..') || path.isAbsolute(relative)) {
    throw new Error(run + ' is outside the project');
  }
  return real;
}

export async function loadCodeTools(
  root: string,
  names: readonly string[],
  agent: string,
): Promise<readonly LoadedCodeTool[]> {
  const wanted = new Set(names);
  const loaded = new Map<string, LoadedCodeTool>();
  for (const file of (await filesBelow(path.join(root, 'tools'))).sort()) {
    const declaration = await defaultExport(file);
    if (!isCodeTool(declaration) || !wanted.has(declaration.name)) continue;
    if (loaded.has(declaration.name)) continue;
    const run = await defaultExport(await runFile(root, declaration.run));
    if (typeof run !== 'function') {
      throw new Error(declaration.run + ' must export a function as its default');
    }
    const input = z.fromJSONSchema(declaration.inputSchema);
    if (!(input instanceof z.ZodObject)) {
      throw new Error(declaration.name + ': its input schema must describe an object');
    }
    const timeoutMs = declaration.limits.timeoutMs;
    loaded.set(declaration.name, {
      name: declaration.name,
      description: declaration.description,
      input,
      call: async (args) =>
        (run as RunFunction)(input.parse(args), {
          signal: AbortSignal.timeout(timeoutMs),
          runId,
          agent,
        }),
    });
  }
  return names.map((name) => {
    const found = loaded.get(name);
    if (found === undefined) throw new Error('code tool ' + name + ' not found under tools/');
    return found;
  });
}

import { createAgent } from "./agent.ts";

const prompt = process.argv.slice(2).join(" ").trim();
if (prompt === "") {
  console.error("Usage: node --env-file=.env src/main.ts \"your prompt\"");
  process.exit(1);
}

let close: (() => Promise<void>) | undefined;
try {
  const created = await createAgent();
  close = created.close;
  const { agent } = created;
  const result = await agent.generate(prompt, { maxSteps: 20 });
  console.log(result.text);
} catch (error) {
  console.error(error instanceof Error ? error.message : String(error));
  process.exitCode = 1;
} finally {
  await close?.();
}
